        function handleImageUpload(event) {

            const file = event.target.files[0];

        // Dynamically crop a high-quality upload to target canvas dimensions (preventing stretching)

        function cropBackground() {

            if (!hasUploadedImage) return;


            const targetW = canvas.width;

            const targetH = canvas.height;


            const tempCanvas = document.createElement('canvas');

            tempCanvas.width = targetW;

            tempCanvas.height = targetH;

            const tempCtx = tempCanvas.getContext('2d');


            const imgAspect = uploadedImage.width / uploadedImage.height;

            const targetAspect = targetW / targetH;

            

            let drawWidth, drawHeight, offsetX, offsetY;


            if (imgAspect > targetAspect) { // Upload is wider than target aspect ratio

                drawHeight = targetH;

                drawWidth = targetH * imgAspect;

                offsetX = -(drawWidth - targetW) / 2;

                offsetY = 0;

            } else { // Upload is taller than target aspect ratio

                drawWidth = targetW;

                drawHeight = targetW / imgAspect;

                offsetX = 0;

                offsetY = -(drawHeight - targetH) / 2;

            }


            tempCtx.drawImage(uploadedImage, offsetX, offsetY, drawWidth, drawHeight);


            activeBackgroundImage.onload = function() {

                drawAsset();

            };

            activeBackgroundImage.src = tempCanvas.toDataURL();

        }

        // Call Gemini Model

        async function callGeminiCopilot(storyText) {

            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

            const systemPrompt = "You are a senior brand strategy copywriter for Arthur Rank Hospice Charity (ARHC). You write deeply moving, clinically dignified, yet highly warm social media posts. Never write cheesy or over-curated sentences. Avoid meme formats. Ensure all quotes feel incredibly authentic and grounded. Return clean JSON containing quotes (under 280 characters, do not put quotation marks in the string), appropriate author names, author job titles, and descriptive AI image generation prompts optimized for background scenic composition.";

            

            const payload = {

                contents: [

                    { parts: [{ text: `Generate content for the pillar: "${currentPillar}". Base the copy on this story description: "${storyText}"` }] }

                ],

                systemInstruction: {

                    parts: [{ text: systemPrompt }]

                },

                generationConfig: {

                    responseMimeType: "application/json",

                    responseSchema: {

                        type: "OBJECT",

                        properties: {

                            quote: { type: "STRING", description: "A heartwarming, high-impact quote or advice statement. Under 280 characters. No literal quotation marks." },

                            author: { type: "STRING", description: "A realistic first name or full name with title." },

                            title: { type: "STRING", description: "A realistic job title, volunteer status, or local company name." },

                            imagePrompt: { type: "STRING", description: "A clean, detailed background photo generation prompt tailored to this theme, focusing on authentic, candid photography style with warm natural lighting." }

                        },

                        required: ["quote", "author", "title", "imagePrompt"]

                    }

                }

            };


            const options = {

                method: "POST",

                headers: { "Content-Type": "application/json" },

                body: JSON.stringify(payload)

            };


            const result = await fetchWithBackoff(url, options);

            const textResponse = result.candidates?.[0]?.content?.parts?.[0]?.text;

            if (textResponse) {

                return JSON.parse(textResponse);

            } else {

                throw new Error("Invalid response from Gemini model");

            }

        }

        // API Caller with Exponential Backoff

        async function callImagenAPI(promptText) {

            const url = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`;

            const payload = {

                instances: [

                    { prompt: promptText }

                ],

                parameters: {

                    sampleCount: 1,

                    aspectRatio: "1:1"

                }

            };


            const options = {

                method: "POST",

                headers: { "Content-Type": "application/json" },

                body: JSON.stringify(payload)

            };


            const result = await fetchWithBackoff(url, options);

            if (result.predictions && result.predictions[0] && result.predictions[0].bytesBase64Encoded) {

                return `data:image/png;base64,${result.predictions[0].bytesBase64Encoded}`;

            } else {

                throw new Error("Invalid API response shape");

            }

        }


        // Robust Fetch backoff implementation matching strategic instructions

        async function fetchWithBackoff(url, options, retries = 5, delay = 1000) {

            try {

                const response = await fetch(url, options);

                if (!response.ok) {

                    throw new Error(`HTTP error! status: ${response.status}`);

                }

                return await response.json();

            } catch (error) {

                if (retries > 0) {

                    await new Promise(resolve => setTimeout(resolve, delay));

                    return fetchWithBackoff(url, options, retries - 1, delay * 2);

                }

                throw error;

            }

        }

        // Custom canvas multi-line text-wrap processor

        function wrapText(context, text, x, y, maxWidth, lineHeight) {

            const words = text.split(' ');

            let line = '';


            for (let n = 0; n < words.length; n++) {

                let testLine = line + words[n] + ' ';

                let metrics = context.measureText(testLine);

                let testWidth = metrics.width;

                if (testWidth > maxWidth && n > 0) {

                    context.fillText(line, x, y);

                    line = words[n] + ' ';

                    y += lineHeight;

                }

                else {

                    line = testLine;

                }

            }

            context.fillText(line, x, y);

        }
